# 实际执行脚本（等价，2026-10-04 编排会话）
import json, glob
ids=[]
for f in sorted(glob.glob('training_data/sfx-v1/annotate-r1/judgments/agy/batch-*.json')):
    d=json.load(open(f,encoding='utf-8'))
    units=d.get('units') or d.get('judgments') or d
    if isinstance(units,dict): units=list(units.values())
    for u in units:
        ids.append(u.get('unit_id') or u.get('id'))
print('total rows:',len(ids))
print('unique ids:',len(set(ids)))
print('dup check:','PASS (no dup)' if len(ids)==len(set(ids)) else 'FAIL')
